alloyapi.py
from openai import OpenAI
import os

client = OpenAI(
  base_url="https://alloyapi.ai.studio/v1",
  api_key=os.environ["ALLOY_API_KEY"],
)

response = client.chat.completions.create(
  model="gpt-4o-mini",
  messages=[{"role": "user", "content": "Hello via AlloyAPI proxy!"}],
  max_tokens=150,
)

print(response.choices[0].message.content)

alloySDK / quickstart.jsNODE
import { Alloy } from 'alloy-sdk';

const alloy = new Alloy({
  defaultProvider: 'vllm',
  cacheTTLMs: 300000
});

const response = await alloy.execute(
  payload => client.chat.completions.create(payload),
  { messages },
  { provider: 'vllm' }
);
